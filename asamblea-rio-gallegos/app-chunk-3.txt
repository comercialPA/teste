    function resetUpload(){state.files=[];$('fileInput').value='';$('previewList').innerHTML='';$('consentCheck').checked=false;$('uploadProgress').classList.remove('show');$('progressBar').style.width='0%';$('progressPercent').textContent='0%';updateUploadButton();}

    function setFiles(files){
      const maxSourceMb=Number(state.bootstrap?.limits?.maxSourceUploadMb||32);
      const maxSourceBytes=maxSourceMb*1024*1024;
      const accepted=[];
      let rejected=0;
      for(const file of Array.from(files||[])){
        if(accepted.length>=40)break;
        const supported=/^image\//.test(file.type)||/\.(jpe?g|png|webp|heic|heif)$/i.test(file.name);
        if(!supported){rejected++;continue;}
        if(file.size>maxSourceBytes){toast(`${file.name}: supera ${maxSourceMb} MB.`,'error');rejected++;continue;}
        accepted.push({file,caption:''});
      }
      state.files=accepted;
      if(rejected&&accepted.length)toast(`${rejected} archivo${rejected===1?'':'s'} no se pudo${rejected===1?'':'ieron'} agregar.`);
      renderPreviews();updateUploadButton();
    }

    function renderPreviews(){
      const list=$('previewList');list.innerHTML='';
      state.files.forEach((item,index)=>{
        const row=document.createElement('div');row.className='preview-row';
        const img=document.createElement('img');img.alt='Vista previa';
        const objectUrl=URL.createObjectURL(item.file);img.src=objectUrl;img.onload=()=>URL.revokeObjectURL(objectUrl);img.onerror=()=>{URL.revokeObjectURL(objectUrl);img.removeAttribute('src');};
        const info=document.createElement('div');
        info.innerHTML=`<strong>${esc(item.file.name)}</strong><span class="hint">${formatBytes(item.file.size)} · hasta 4K</span><input maxlength="220" placeholder="Leyenda opcional para esta foto">`;
        info.querySelector('input').addEventListener('input',e=>state.files[index].caption=e.target.value);
        row.append(img,info);list.appendChild(row);
      });
    }

    function updateUploadButton(){$('btnStartUpload').disabled=!(state.files.length&&$('consentCheck').checked&&state.profile);}
