    function downloadZip(){
      const ids=Array.from(state.selection);
      if(!ids.length){toast('Seleccioná al menos una foto.');return;}
      const max=Number(state.bootstrap?.limits?.maxZipPhotos||30);
      if(ids.length>max){toast(`Podés descargar hasta ${max} fotos por ZIP.`,'error');return;}
      const btn=$('btnZip');btn.disabled=true;btn.textContent='Preparando…';
      server('apiCreateZip',ids).then(r=>{if(r?.url)window.open(r.url,'_blank');toast(`ZIP listo con ${r.count} fotos.`,'success');}).catch(e=>toast(e.message,'error')).finally(()=>{btn.disabled=false;btn.textContent='Descargar ZIP';});
    }

    function resetUpload(){state.files=[];$('fileInput').value='';$('previewList').innerHTML='';$('consentCheck').checked=false;$('uploadProgress').classList.remove('show');$('progressBar').style.width='0%';$('progressPercent').textContent='0%';updateUploadButton();}

    function setFiles(files){
      state.files=(files||[]).filter(f=>/^image\//.test(f.type)||/\.(jpe?g|png|webp|heic|heif)$/i.test(f.name)).slice(0,40).map(f=>({file:f,caption:''}));
      renderPreviews();updateUploadButton();
    }

    function renderPreviews(){
      const list=$('previewList');list.innerHTML='';
      state.files.forEach((item,index)=>{
        const row=document.createElement('div');row.className='preview-row';
        const img=document.createElement('img');img.alt='Vista previa';
        const reader=new FileReader();reader.onload=e=>img.src=e.target.result;reader.readAsDataURL(item.file);
        const info=document.createElement('div');
        info.innerHTML=`<strong>${esc(item.file.name)}</strong><span class="hint">${formatBytes(item.file.size)}</span><input maxlength="220" placeholder="Leyenda opcional para esta foto">`;
        info.querySelector('input').addEventListener('input',e=>state.files[index].caption=e.target.value);
        row.append(img,info);list.appendChild(row);
      });
    }
