    function getAppShareUrl(){
      const fromBootstrap=String(state.bootstrap?.appUrl||'').trim();
      if(fromBootstrap)return fromBootstrap.split('?')[0].split('#')[0];
      const current=String(location.href||'').split('?')[0].split('#')[0];
      return current;
    }

    function openAppShare(){
      const url=getAppShareUrl();
      $('appShareUrl').value=url;
      openModal('shareAppModal');
      setTimeout(renderAppQr,30);
    }

    function renderAppQr(){
      const host=$('appQrCode');
      if(!host)return;
      const url=getAppShareUrl();
      $('appShareUrl').value=url;
      host.innerHTML='';
      if(!url){host.innerHTML='<div class="qr-fallback">No pudimos identificar el enlace público del app.</div>';return;}
      if(typeof QRCode==='undefined'){
        host.innerHTML='<div class="qr-fallback">El generador QR no pudo cargarse. Podés usar el botón “Copiar link”.</div>';
        return;
      }
      try{
        new QRCode(host,{text:url,width:260,height:260,colorDark:'#08151d',colorLight:'#ffffff',correctLevel:QRCode.CorrectLevel.M});
      }catch(e){host.innerHTML='<div class="qr-fallback">No pudimos generar el QR. Podés copiar el enlace debajo.</div>';}
    }

    async function copyAppLink(){
      const url=getAppShareUrl();
      if(!url){toast('No pudimos identificar el enlace del app.','error');return;}
      await copyText(url);
    }

    async function shareApp(){
      const url=getAppShareUrl();
      if(!url){toast('No pudimos identificar el enlace del app.','error');return;}
      const data={title:'Recuerdos · Asamblea Río Gallegos 2026',text:'Galería de recuerdos de la Asamblea Río Gallegos 2026',url};
      try{
        if(navigator.share)await navigator.share(data);
        else await copyText(url);
      }catch(e){if(e.name!=='AbortError')toast('No pudimos compartir el app.','error');}
    }

    async function sharePhoto(photo){
      if(!photo)return;const base=state.bootstrap?.appUrl||location.href.split('?')[0];const url=`${base}?foto=${encodeURIComponent(photo.id)}`;
      const data={title:'Asamblea Río Gallegos 2026',text:`Foto compartida por ${photo.uploaderName} · ${photo.congregation}`,url};
      try{if(navigator.share)await navigator.share(data);else{await copyText(url);toast('Enlace copiado.','success');}}catch(e){if(e.name!=='AbortError')toast('No pudimos compartir el enlace.','error');}
    }
