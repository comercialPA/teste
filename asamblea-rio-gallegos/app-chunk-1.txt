    const LS_PROFILE='rg26_profile_v1';
    const LS_FAVORITES='rg26_favorites_v1';
    const SS_ADMIN='rg26_admin_key_v1';
    const state={
      bootstrap:null,
      profile:readJson(LS_PROFILE),
      favorites:new Set(readJson(LS_FAVORITES)||[]),

    function boot(){
      bindEvents();
      renderSkeletons();
      server('apiBootstrap', state.profile?.profileId||'')
        .then(data=>{
          state.bootstrap=data;
          $('eventDateLabel').textContent=data.event?.dateLabel||formatEventDate(data.event?.date);
          $('disclaimer').textContent=data.event.disclaimer||$('disclaimer').textContent;
          $('statPhotos').textContent=numberFmt(data.stats.photos||0);
          $('statCongregations').textContent=numberFmt(data.stats.congregations||0);
          fillCongregations(data.congregations||[]);
          applyGalleryResult(data.gallery,true);
          refreshProfileUi();
          resolveDeepLink();
        })
        .catch(err=>{ clearGallery(); toast(err.message||String(err),'error'); });
    }
