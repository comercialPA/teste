    async function prepareImage(file,maxMb){
      // Vercel Functions limita el cuerpo HTTP. Cada foto viaja por debajo de 2,5 MB
      // antes de convertirse a base64; la reducción solo ocurre cuando hace falta.
      const transportMb=Math.min(Number(maxMb||12),2.5);
      const maxBytes=transportMb*1024*1024;
      if(file.size<=maxBytes)return {file,name:file.name,type:file.type||guessMime(file.name)};
      if(!/^image\/(jpeg|jpg|png|webp)$/i.test(file.type))throw new Error(`supera ${transportMb} MB para la carga web`);
      try{
        const bitmap=await createImageBitmap(file);
        const maxSide=3000,scale=Math.min(1,maxSide/Math.max(bitmap.width,bitmap.height));
        const canvas=document.createElement('canvas');canvas.width=Math.round(bitmap.width*scale);canvas.height=Math.round(bitmap.height*scale);
        canvas.getContext('2d',{alpha:false}).drawImage(bitmap,0,0,canvas.width,canvas.height);bitmap.close?.();
        let quality=.90,blob=await canvasBlob(canvas,'image/jpeg',quality);
        while(blob.size>maxBytes&&quality>.62){quality-=.07;blob=await canvasBlob(canvas,'image/jpeg',quality);}
        if(blob.size>maxBytes)throw new Error(`sigue superando ${transportMb} MB después de optimizarla`);
        return {file:blob,name:file.name.replace(/\.[^.]+$/,'.jpg'),type:'image/jpeg'};
      }catch(e){throw new Error(`supera ${transportMb} MB y no pudimos optimizarla`);}
    }

    function canvasBlob(canvas,type,quality){return new Promise((res,rej)=>canvas.toBlob(b=>b?res(b):rej(new Error('No se pudo optimizar')),type,quality));}
    function readFile(file){return new Promise((res,rej)=>{const r=new FileReader();r.onload=()=>res(r.result);r.onerror=()=>rej(new Error('No se pudo leer la foto'));r.readAsDataURL(file);});}
